```ts
import rss from "@astrojs/rss";
import { getCollection } from "astro:content";

export async function GET(context) {
  const blog = await getCollection("blog");
  const posts = blog
    .filter((p) => !p.data.draft)
    .sort((a, b) => (b.data.date?.getTime() ?? 0) - (a.data.date?.getTime() ?? 0));

  return rss({
    title: "亦幸小阁",
    description: "Hencter Lew 的技术博客",
    site: context.site,
    items: posts.map((post) => ({
      title: post.data.title,
      description: post.data.description,
      pubDate: post.data.date,
      link: `/${post.id}`,
    })),
  });
}
```

## 5. hreflang (国际化)

### 5.1 当前状态: 🔴 完全缺失

全站 0 个 hreflang 标签。

- Astro i18n 路由已配置（`zh-CN` = `/`, `en-US` = `/en`）
- 页面间有语言切换链接（BaseLayout header）
- 但 `<head>` 中无任何 `<link rel="alternate" hreflang="...">`

### 5.2 影响
搜索引擎可能将中英文相同路径的页面视为**重复内容**，导致：
- 错误的语言版本被索引
- 排名信号被稀释
- i18n 投入的 SEO 价值为零

### 5.3 修复方案

在 `Meta.astro` 中添加 hreflang 支持：

```astro
---
interface Props {
  // ... existing props
  hreflang?: { lang: string; href: string }[];
  hreflangDefault?: string;
}

const { hreflang = [], hreflangDefault } = Astro.props;
---

<!-- 在 canonical 后面添加 -->
{hreflang.map(({ lang, href }) => (
  <link rel="alternate" hreflang={lang} href={href} />
))}
{hreflangDefault && (
  <link rel="alternate" hreflang="x-default" href={hreflangDefault} />
)}
```

然后在 `BaseLayout.astro` 中传入：
```astro
<Meta
  hreflang={[
    { lang: "zh-CN", href: `https://hencte.top${zhPath}` },
    { lang: "en-US", href: `https://hencte.top/en${enPath}` },
  ]}
  hreflangDefault="https://hencte.top/"
/>
```

|------|-----------|-----------|:---:|
| 首页 | `/` | `/en` | ❌ |

## 10. 小说内容保护策略评估

### 当前措施
1. `NovelLayout.astro:40`: `<meta name="robots" content="noai, noimageai, noarchive, nocache" />`
2. HTML 注释声明版权
3. 客户端 JS: 禁止右键/选择/复制/DevTools 检测

### 问题
- `noai` 和 `noimageai` **不是标准 robots meta 值**。搜索引擎不认识。
- 正确的 AI 爬虫拦截应在 **robots.txt** 中按 User-Agent 声明（见 1.1 节的修复方案）。
- `noarchive, nocache` 可保留，建议加上 `noindex`（如果确实不希望小说被索引）。但用户意图可能是**允许索引但禁止 AI 训练**——这是一个微妙的需求。

### 推荐策略

**方案 A: 小说允许索引但禁止 AI 训练**（当前设计意图）
```
robots.txt:
  User-agent: GPTBot        → Disallow: /novel/
  User-agent: anthropic-ai  → Disallow: /novel/
  User-agent: Google-Extended → Disallow: /novel/
  User-agent: CCBot         → Disallow: /novel/

NovelLayout head:
  <meta name="robots" content="noarchive, nocache" />
  （移除 noai, noimageai）
```

**方案 B: 小说完全不可索引**
```
NovelLayout head:
  <meta name="robots" content="noindex, noarchive, nocache" />
```

> 当前用户意图似乎是方案 A，因为小说内容希望被读者发现（通过站点导航），但不希望被 AI 公司用于训练。
